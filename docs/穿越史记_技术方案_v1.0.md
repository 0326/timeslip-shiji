## 1. 技术选型总览

### 核心技术栈

| 层级 | 技术 | 版本 | 选型理由 |
|------|------|------|---------|
| 框架 | React | 18.x | 生态成熟，组件复用，状态管理方案丰富 |
| 构建工具 | Vite | 5.x | 热更新快，ESM原生支持，比CRA轻量 |
| 语言 | TypeScript | 5.x | 类型安全，数据模型复杂场景必要 |
| 叙事引擎 | inkjs | 2.x | Ink语言的JS运行时，专为分支叙事设计 |
| 状态管理 | Zustand | 4.x | 轻量，无样板代码，适合中型应用 |
| 路由 | React Router | 6.x | 标准方案 |
| 可视化 | D3.js | 7.x | 人物关系图、时间轴 |
| 样式 | Tailwind CSS + CSS Modules | 3.x | 原子类快速布局，模块化避免污染 |
| 动效 | Framer Motion | 11.x | React生态最佳动效库，声明式API |
| 图标 | Lucide React | latest | 轻量，Tree-shakable |
| HTTP | 无（纯前端）/ Fetch API | - | Demo阶段无后端，数据本地JSON |

### 叙事引擎选型说明：为什么是 inkjs 而非 Ren'Py

```
Ren'Py
  ✗ 无法嵌入Web页面
  ✗ UI完全由引擎控制，无法自定义水墨风格
  ✗ 无法与D3/React生态集成
  ✗ WebAssembly导出包体 >30MB

inkjs
  ✓ npm install 即用，bundle size ~150KB
  ✓ 叙事逻辑与UI完全解耦
  ✓ 原生JS，可与任意Web技术栈集成
  ✓ 支持变量、条件、函数，足以实现死亡判定和成就触发
  ✓ .ink文件由内容编辑者维护，与工程代码分离
```

### inkjs 运行时封装

```typescript
// src/engine/inkRunner.ts
import { Story } from 'inkjs'

export interface InkState {
  text: string[]
  choices: InkChoice[]
  tags: Record<string, string>
  isDead: boolean
  isEnded: boolean
  variables: Record<string, unknown>
}

export interface InkChoice {
  index: number
  text: string
}

export class InkRunner {
  private story: Story
  private onAchievement?: (id: string) => void

  constructor(inkJson: object, callbacks?: {
    onAchievement?: (id: string) => void
    onBGM?: (track: string) => void
    onCharacter?: (id: string, expr: string, pos: string) => void
    onBackground?: (bg: string) => void
  }) {
    this.story = new Story(inkJson)
    this.onAchievement = callbacks?.onAchievement

    // 注册外部函数供.ink调用
    this.story.BindExternalFunction('triggerAchievement', (id: string) => {
      callbacks?.onAchievement?.(id)
    })
    this.story.BindExternalFunction('playBGM', (track: string) => {
      callbacks?.onBGM?.(track)
    })
    this.story.BindExternalFunction('showCharacter', (
      id: string, expr: string, pos: string
    ) => {
      callbacks?.onCharacter?.(id, expr, pos)
    })
    this.story.BindExternalFunction('changeBackground', (bg: string) => {
      callbacks?.onBackground?.(bg)
    })
  }

  // 推进故事，收集所有文本直到需要选择或结束
  advance(): InkState {
    const textLines: string[] = []
    const tags: Record<string, string> = {}

    while (this.story.canContinue) {
      const line = this.story.Continue()
      if (line.trim()) textLines.push(line.trim())

      // 解析标签
      for (const tag of this.story.currentTags) {
        const [key, ...rest] = tag.split(':')
        tags[key.trim()] = rest.join(':').trim()
      }

      // 遇到死亡标签立即停止收集
      if (tags['DEATH'] === 'true') break
    }

    return {
      text: textLines,
      choices: this.story.currentChoices.map((c, i) => ({
        index: i,
        text: c.text
      })),
      tags,
      isDead: tags['DEATH'] === 'true',
      isEnded: !this.story.canContinue && this.story.currentChoices.length === 0,
      variables: this.getVariables()
    }
  }

  // 做出选择
  choose(choiceIndex: number): InkState {
    this.story.ChooseChoiceIndex(choiceIndex)
    return this.advance()
  }

  // 获取当前所有变量值（用于成就检测）
  getVariables(): Record<string, unknown> {
    return {
      death_count: this.story.variablesState['death_count'],
      correct_choices: this.story.variablesState['correct_choices'],
      fanzen_ignored: this.story.variablesState['fanzen_ignored'],
    }
  }

  // 序列化存档
  getSaveState(): string {
    return this.story.state.ToJson()
  }

  // 从存档恢复
  loadSaveState(json: string): void {
    this.story.state.LoadJson(json)
  }
}
```

### 核心游戏Hook

```typescript
// src/hooks/useInkStory.ts
import { useState, useEffect, useCallback, useRef } from 'react'
import { InkRunner, InkState } from '../engine/inkRunner'
import { useAchievement } from './useAchievement'
import { useSaveManager } from './useSaveManager'

export function useInkStory(storylineId: string, characterId: string) {
  const [state, setState] = useState<InkState | null>(null)
  const [isLoading, setIsLoading] = useState(true)
  const [sceneState, setSceneState] = useState({
    background: 'default',
    characters: {} as Record<string, { expression: string; position: string }>,
    speaker: '',
  })
  const runnerRef = useRef<InkRunner | null>(null)
  const { triggerAchievement } = useAchievement()
  const { saveProgress } = useSaveManager()

  useEffect(() => {
    async function loadStory() {
      setIsLoading(true)
      try {
        // 动态加载对应的ink编译结果
        const inkJson = await import(
          `../ink/${storylineId}/${characterId}.json`
        )
        runnerRef.current = new InkRunner(inkJson.default, {
          onAchievement: triggerAchievement,
          onBGM: (track) => { /* 播放BGM */ },
          onCharacter: (id, expr, pos) => {
            setSceneState(prev => ({
              ...prev,
              characters: { ...prev.characters, [id]: { expression: expr, position: pos } }
            }))
          },
          onBackground: (bg) => {
            setSceneState(prev => ({ ...prev, background: bg }))
          },
        })
        const initialState = runnerRef.current.advance()
        setState(initialState)
      } finally {
        setIsLoading(false)
      }
    }
    loadStory()
  }, [storylineId, characterId])

  const makeChoice = useCallback((choiceIndex: number) => {
    if (!runnerRef.current) return
    const newState = runnerRef.current.choose(choiceIndex)
    setState(newState)
    // 自动存档
    saveProgress(storylineId, characterId, runnerRef.current.getSaveState())
  }, [storylineId, characterId])

  const advance = useCallback(() => {
    if (!runnerRef.current) return
    const newState = runnerRef.current.advance()
    setState(newState)
  }, [])

  return { state, sceneState, isLoading, makeChoice, advance }
}
```

```typescript
import { useState, useEffect } from 'react'
import { motion } from 'framer-motion'
import { usePlayStore } from '../../store/playStore'

const SPEED_MAP = { slow: 60, normal: 30, fast: 10 }  // ms per char

export default function DialogueBox({
  lines, speaker
}: { lines: string[]; speaker: string }) {
  const { textSpeed } = usePlayStore()
  const [displayText, setDisplayText] = useState('')
  const [isComplete, setIsComplete] = useState(false)
  const fullText = lines.join('\n')

  useEffect(() => {
    setDisplayText('')
    setIsComplete(false)
    let i = 0
    const interval = setInterval(() => {
      if (i >= fullText.length) {
        setIsComplete(true)
        clearInterval(interval)
        return
      }
      setDisplayText(fullText.slice(0, ++i))
    }, SPEED_MAP[textSpeed])
    return () => clearInterval(interval)
  }, [fullText, textSpeed])

  // 点击立即完成
  const skipToEnd = () => {
    setDisplayText(fullText)
    setIsComplete(true)
  }

  return (
    <motion.div
      className="dialogue-box"
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      onClick={!isComplete ? skipToEnd : undefined}
    >
      {speaker && (
        <div className="speaker-name">{speaker}</div>
      )}
      <div className="dialogue-text">
        {displayText}
        {!isComplete && <span className="cursor">|</span>}
      </div>
      {isComplete && (
        <div className="continue-hint">▼ 点击继续</div>
      )}
    </motion.div>
  )
}
```

### 抽卡逻辑

```typescript
// src/hooks/useGacha.ts
import { useUserStore } from '../store/userStore'
import { CURRENT_POOL } from '../data/gachaPools'

export function useGacha() {
  const { progress, spendGachaTicket, unlockCharacter, addFragment } = useUserStore()

  function pull(count: 1 | 10): GachaPullResult[] {
    if (!spendGachaTicket(count)) {
      throw new Error('抽卡券不足')
    }

    const results: GachaPullResult[] = []
    let pity = progress.gacha.pityCount

    for (let i = 0; i < count; i++) {
      pity++
      const character = drawCharacter(pity, progress.gacha.ownedCharacters)
      const isDuplicate = progress.gacha.ownedCharacters.includes(character.id)

      if (isDuplicate) {
        addFragment(CURRENT_POOL.rules.fragmentsOnDuplicate)
      } else {
        unlockCharacter(character.id)
      }

      results.push({ character, isDuplicate, pity })

      // 保底重置
      if (pity >= CURRENT_POOL.rules.pityAt) pity = 0
    }

    return results
  }

  function drawCharacter(pity: number, owned: string[]) {
    const pool = CURRENT_POOL.characters
    const unowned = pool.filter(c => !owned.includes(c.characterId))

    // 达到保底，从未拥有角色中选
    if (pity >= CURRENT_POOL.rules.pityAt && unowned.length > 0) {
      return weightedRandom(unowned)
    }

    return weightedRandom(pool)
  }

  function weightedRandom(chars: GachaPoolCharacter[]) {
    const total = chars.reduce((sum, c) => sum + c.weight, 0)
    let rand = Math.random() * total
    for (const char of chars) {
      rand -= char.weight
      if (rand <= 0) return char
    }
    return chars[chars.length - 1]
  }

  return { pull }
}
```

```typescript
  const svgRef = useRef<SVGSVGElement>(null)

  useEffect(() => {
    if (!svgRef.current) return
    const svg = d3.select(svgRef.current)
    svg.selectAll('*').remove()

    const W = svgRef.current.clientWidth
    const H = svgRef.current.clientHeight

      })

      source: mainCharacter.id,
      target: r.targetId,
      type: r.type,
      label: r.label,
    }))

    // 力导向模拟
    const simulation = d3.forceSimulation(nodes)
      .force('link', d3.forceLink(links).id((d: any) => d.id).distance(120))
      .force('charge', d3.forceManyBody().strength(-300))
      .force('center', d3.forceCenter(W / 2, H / 2))

    const g = svg.append('g')

    // 边
    const link = g.append('g')
      .selectAll('line')
      .data(links)
      .join('line')
      .attr('stroke', d => LINK_COLORS[d.type])
      .attr('stroke-width', 1.5)
      .attr('stroke-opacity', 0.6)

    // 边标签
    const linkLabel = g.append('g')
      .selectAll('text')
      .data(links)
      .join('text')
      .text(d => d.label)
      .attr('font-size', '10px')

      .attr('stroke-width', d => d.isMain ? 2 : 1)
      .call(
        d3.drag<SVGCircleElement, Node>()
          .on('start', (event, d) => {
            if (!event.active) simulation.alphaTarget(0.3).restart()
            d.fx = d.x; d.fy = d.y
          })
          .on('drag', (event, d) => { d.fx = event.x; d.fy = event.y })
          .on('end', (event, d) => {
            if (!event.active) simulation.alphaTarget(0)
            d.fx = null; d.fy = null

    // 节点文字

      label
        .attr('x', d => d.x!)
        .attr('y', d => d.y!)
    })

    return () => simulation.stop()
  }, [mainCharacter])

  return <svg ref={svgRef} width="100%" height="400" />
```

## 12. 存档与本地持久化

### 存档分层策略

```
Layer 1: Zustand + persist（自动）
  → 用户进度、积分、抽卡记录、成就
  → localStorage: 'shiji-user-progress'
  → 每次store变更自动同步

Layer 2: inkjs 存档（手动触发）
  → 每个故事视角的游戏进度
  → localStorage: 'shiji-ink-{storylineId}-{characterId}'
  → 在节点完成时、选择后自动保存

Layer 3: 未来后端同步（MVP后）
  → 用户注册后云端备份
  → 多设备同步
```

```typescript
// src/engine/saveManager.ts
const SAVE_KEY_PREFIX = 'shiji-ink-'

export class SaveManager {
  static save(storylineId: string, characterId: string, inkState: string): void {
    const key = `${SAVE_KEY_PREFIX}${storylineId}-${characterId}`
    localStorage.setItem(key, inkState)
  }

  static load(storylineId: string, characterId: string): string | null {
    const key = `${SAVE_KEY_PREFIX}${storylineId}-${characterId}`
    return localStorage.getItem(key)
  }

  static hasSave(storylineId: string, characterId: string): boolean {
    return !!this.load(storylineId, characterId)
  }

  static deleteSave(storylineId: string, characterId: string): void {
    const key = `${SAVE_KEY_PREFIX}${storylineId}-${characterId}`
    localStorage.removeItem(key)
  }

  // 导出所有存档（未来分享/备份功能）
  static exportAll(): string {
    const saves: Record<string, string> = {}
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i)!
      if (key.startsWith(SAVE_KEY_PREFIX)) {
        saves[key] = localStorage.getItem(key)!
      }
    }
    return JSON.stringify(saves)
  }
}
```

```typescript
// 用Canvas绘制，生成PNG分享图

export function useShareCard() {
  async function generateAchievementCard(params: {

    ctx.textAlign = 'center'
    ctx.fillText('穿越·史记', 300, 70)

    // 成就名称
    ctx.fillStyle = '#F2ECD8'
    ctx.font = 'bold 36px "Noto Serif SC", serif'
    ctx.fillText(params.achievementName, 300, 200)

    // 原文
    ctx.fillStyle = 'rgba(242,236,216,0.6)'
    ctx.font = '18px "Noto Serif SC", serif'
    const lines = wrapText(ctx, params.classicalQuote, 500)
    lines.forEach((line, i) => {
      ctx.fillText(line, 300, 300 + i * 30)
    })

    // 角色名和用户名
    ctx.fillStyle = '#B8973A'
    ctx.font = '16px "Noto Sans SC", sans-serif'
    ctx.fillText(`${params.userName} 穿越成为 ${params.characterName}`, 300, 800)

    return new Promise(resolve => {
      canvas.toBlob(blob => resolve(blob!), 'image/png')
    })
  }

  function wrapText(ctx: CanvasRenderingContext2D, text: string, maxWidth: number) {
    const words = text.split('')
    const lines: string[] = []
    let current = ''
    for (const char of words) {
      const test = current + char
      if (ctx.measureText(test).width > maxWidth) {
        lines.push(current)
        current = char
      } else {
        current = test
      }
    }
    if (current) lines.push(current)
    return lines
  }

  return { generateAchievementCard }
}
```
